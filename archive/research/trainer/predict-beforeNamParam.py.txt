# Custom logic to check mutually inclusive options
if not ((args.task == 'regression' and args.Target_num) or (args.task == 'classification' and args.Class_num)):

# Load the model    
if os.path.isfile(args.modelpath):
    print("=> loading model params '{}'".format(args.modelpath))
    model_checkpoint = torch.load(args.modelpath,

def main():
    global args, model_args, date_time_str

    # load data
    dataset = CIFData(*args.data_options,*args.zipfile,args.id_prop,*args.use_features,args.useable_samples,
                      task=args.task,targets = args.Target_num
                      if args.Target_num else args.Class_num)
    collate_fn = collate_pool
    test_loader = DataLoader(dataset, batch_size=args.batch_size, shuffle=True,
                             num_workers=args.workers, collate_fn=collate_fn,
                             pin_memory=args.cuda)
    
    
    # build model
    structures, _, _ = dataset[0]
    orig_atom_fea_len = structures[0].shape[-1]
    nbr_fea_len = structures[1].shape[-1]
    model = CrystalGraphConvNet_1(orig_atom_fea_len, nbr_fea_len,
                                atom_fea_len=model_args.atom_fea_len,
                                added_fea_len=model_args.Additional_feat_num,
                                classnum = model_args.Class_num if model_args.task =='classification' else model_args.Target_num,
                                n_conv=model_args.n_conv,
                                h_fea_len=model_args.h_fea_len,
                                n_h=model_args.n_h,
                                classification=True if model_args.task =='classification' else False)
    
    if args.cuda:
        model.cuda()
    
    #normalizer = Normalizer(torch.zeros(3))

    sample_data_list = [dataset[1]]
    _, sample_target, _ = collate_pool(sample_data_list)

    # obtain target value normalizer
    if args.task == 'classification':
        normalizer = Normalizer(sample_target[:, :-1])

    else:
        normalizer = Normalizer(sample_target)
        target_normalizer = Normalizer(sample_target[:,-args.Target_num:])
        
        regNorm = {'normalizer':normalizer,
                   'target_normalizer':target_normalizer}

    # optionally resume from a checkpoint
    if os.path.isfile(args.modelpath):
        print("=> loading model '{}'".format(args.modelpath))
        checkpoint = torch.load(args.modelpath,
                                map_location=lambda storage, loc: storage)
        model.load_state_dict(checkpoint['state_dict'])
        if args.task == 'classification':
                normalizer.load_state_dict(checkpoint['normalizer'])
        else:
                regNorm['normalizer'].load_state_dict(checkpoint['normalizer'])
                checkpoint['normalizer']['mean'] = checkpoint['normalizer']['mean'][-args.Target_num:]
                checkpoint['normalizer']['std'] = checkpoint['normalizer']['std'][-args.Target_num:] 
                regNorm['target_normalizer'].load_state_dict(checkpoint['normalizer'])

        print("=> loaded model '{}' (epoch {}, validation {})"
              .format(args.modelpath, checkpoint['epoch'],
                      checkpoint['best_mae_error']))
    else:
        print("=> no model found at '{}'".format(args.modelpath))

     # Get the current date and time
    now = datetime.datetime.now()
    # Format the date and time as a string
    date_time_str = now.strftime("%Y%m%d_%H%M%S")

    if args.task == 'regression':
        validate(test_loader, model, regNorm, test=True)
    else:
        validate(test_loader, model, normalizer, test=True)




            
def validate(val_loader, model, normalizer,test=False,**kwargs):
        
        batch_time = AverageMeter()
            
        if test or ('epoch' in kwargs and args.epochs % (kwargs['epoch']+1) == 0):
            test_targets = []
            test_preds = []
            test_probs = []
            crys_encodes = []
            test_cif_ids = []

        # switch to evaluate mode
        model.eval()

        end = time.time()
        for i, (input, target, batch_cif_ids) in enumerate(val_loader):
            if args.cuda:
                with torch.no_grad():
                    input_var = (Variable(input[0].cuda(non_blocking=True)),
                                Variable(input[1].cuda(non_blocking=True)),
                                input[2].cuda(non_blocking=True),
                                [crys_idx.cuda(non_blocking=True) for crys_idx in input[3]])
            else:
                with torch.no_grad():
                    input_var = (Variable(input[0]),
                                Variable(input[1]),
                                input[2],
                                input[3])
                    
            if args.task == 'regression':
                target_normed = normalizer['normalizer'].norm(target)
            else:
                target_normed = target[:,-1].view(-1).long()
                additional_features = target[:, :-1]
                additional_features = normalizer.norm(additional_features)

            if args.cuda:
                if args.task == 'regression':
                    additional_features = target_normed[:, :-args.Target_num].cuda(non_blocking=True)
                    target_var = Variable(target_normed[:,-args.Target_num:].cuda(non_blocking=True))
                else:
                    target_var = Variable(target_normed.cuda(non_blocking=True))
                    target = target_normed.cuda(non_blocking=True)

            else:
                if args.task == 'regression':
                    additional_features = target_normed[:, :-args.Target_num]
                    target_var = Variable(target_normed[:,-args.Target_num:])
                else:
                    target = target_normed
                    target_var = Variable(target_normed)

            # compute output
            output,enc_crys_fea  = model(*input_var,additional_features.cuda() if args.cuda else additional_features)
            

            # Gather the model predictions
            if args.task == 'regression':
                if test:
                    test_pred = normalizer['target_normalizer'].denorm(output.data.cpu())
                    test_target = target[:,-args.Target_num:]
                    test_preds += test_pred.tolist()
                    test_targets += test_target.tolist()
                    crys_encodes += enc_crys_fea.tolist()
                    test_cif_ids += batch_cif_ids
            
            if args.task == 'classification':    
                if test:
                        test_pred = torch.exp(output.data.cpu())
                        test_prob = test_pred
                        test_target = target_var
                        assert test_pred.shape[1] == args.Class_num 
                        test_pred = torch.argmax(test_pred,axis=1)
                        test_preds += test_pred[:].tolist()
                        test_probs += test_prob.tolist()
                        test_targets += test_target.view(-1).tolist()
                        crys_encodes += enc_crys_fea.tolist() 
                        test_cif_ids += batch_cif_ids

            # measure elapsed time
            batch_time.update(time.time() - end)
            end = time.time()

            if i % args.print_freq == 0:
                if args.task == 'regression':
                    print('Test: [{0}/{1}]\t'
                        'Time {batch_time.val:.3f} ({batch_time.avg:.3f})'.format(
                        i, len(val_loader), batch_time=batch_time))
                else:
                    print('Test: [{0}/{1}]\t'
                        'Time {batch_time.val:.3f} ({batch_time.avg:.3f})'.format(
                        i, len(val_loader), batch_time=batch_time))


        if test:
            import csv
            with open(f'predicted_results_{date_time_str}.csv', 'w') as f:
                writer = csv.writer(f)
                if args.task == 'classification':
                    for cif_id, target, pred, probs, encodes in zip(test_cif_ids, test_targets,
                                                test_preds,test_probs, crys_encodes):
                        writer.writerow((cif_id, target, pred,probs,encodes))
                else:
                    for cif_id, target, pred, encodes in zip(test_cif_ids, test_targets,
                                                test_preds, crys_encodes):
                        writer.writerow((cif_id, target, pred,encodes))

class AverageMeter(object):
    """Computes and stores the average and current value"""

    def __init__(self):
        self.reset()

    def reset(self):
        self.val = 0
        self.avg = 0
        self.sum = 0
        self.count = 0

    def update(self, val, n=1):
        self.val = val
        self.sum += val * n
        self.count += n
        self.avg = self.sum / self.count

    def numpy(self):
        self.avg = self.avg.numpy()

class Normalizer(object):
    """Normalize a Tensor column by column and restore it later. """

    def __init__(self, tensor):
        """tensor is taken as a sample to calculate the mean and std"""
        self.mean = torch.mean(tensor, dim=0)
        self.std = torch.std(tensor, dim=0)

    def norm(self, tensor):
        return (tensor - self.mean) / self.std

    def denorm(self, normed_tensor):
        return normed_tensor * self.std + self.mean

    def state_dict(self):
        return {'mean': self.mean,
                'std': self.std}

    def load_state_dict(self, state_dict):
        self.mean = state_dict['mean']
        self.std = state_dict['std']

if __name__ == '__main__':
    main()
