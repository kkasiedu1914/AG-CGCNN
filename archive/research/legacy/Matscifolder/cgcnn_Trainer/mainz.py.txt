def train(train_loader, model, criterion, optimizer, epoch, normalizer,**kwargs):
        batch_time = AverageMeter()
        data_time = AverageMeter()
        losses = AverageMeter()
        if 'df_train' in kwargs:
            logs = []
        if args.task == 'regression':
            mae_errors = AverageMeter()
        else:
            accuracies = AverageMeter()
            precisions = AverageMeter()
            recalls = AverageMeter()
            fscores = AverageMeter()
            

        # switch to train mode
        model.train()

        end = time.time()
        for i, (input, target, _) in enumerate(train_loader):
            data_time.update(time.time() - end)

            if args.cuda:
                input_var = (Variable(input[0].cuda(non_blocking=True)),
                            Variable(input[1].cuda(non_blocking=True)),
                            input[2].cuda(non_blocking=True),
                            [crys_idx.cuda(non_blocking=True) for crys_idx in input[3]])
            else:
                input_var = (Variable(input[0]),
                            Variable(input[1]),
                            input[2],
                            input[3])
                
            # normalize target and features
            if args.task == 'regression':
                target_normed = normalizer['normalizer'].log_scale(target)
            else:
                target_normed = target[:,-1].view(-1).long()  # Do not change it 
                additional_features = target[:, :-1]
                additional_features = normalizer.norm(additional_features)
            
            if args.cuda:
                if args.task == 'regression':
                    additional_features = target_normed[:, :-args.Target_num].cuda(non_blocking=True)
                    target_var = Variable(target_normed[:,-args.Target_num:].cuda(non_blocking=True))
                else:
                    target_var = Variable(target_normed.cuda(non_blocking=True))
                    target = target_normed.cuda(non_blocking=True)

            else:
                if args.task == 'regression':
                    additional_features = target_normed[:, :-args.Target_num]
                    target_var = Variable(target_normed[:,-args.Target_num:])

                else:
                    target = target_normed # Do not do this regression. This is correct for classiication (last column)
                    target_var = Variable(target_normed)



            # compute output
            output = model(*input_var,additional_features.cuda() if args.cuda else additional_features)
            loss = criterion(output, target_var)

            # measure accuracy and record loss
            if args.task == 'regression':
                mae_error = mae(normalizer['target_normalizer'].log_descale(output.data.cpu()), target[:,-args.Target_num:])
                losses.update(loss.data.cpu(), target.size(0))
                mae_errors.update(mae_error, target.size(0))
            else:
                accuracy, precision, recall, fscore = \
                    class_eval(output.data.cpu(), target)
                losses.update(loss.data.cpu().item(), target.size(0))
                accuracies.update(accuracy, target.size(0))
                precisions.update(precision, target.size(0))
                recalls.update(recall, target.size(0))
                fscores.update(fscore, target.size(0))
                
            # compute gradient and do SGD step
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            # measure elapsed time
            batch_time.update(time.time() - end)
            end = time.time()
            
            if i % args.print_freq == 0:
                if args.task == 'regression':
                    print('Epoch: [{0}][{1}/{2}]\t'
                        'Time {batch_time.val:.3f} ({batch_time.avg:.3f})\t'
                        'Data {data_time.val:.3f} ({data_time.avg:.3f})\t'
                        'Loss {loss.val:.4f} ({loss.avg:.4f})\t'
                        'MAE {mae_errors.val:.3f} ({mae_errors.avg:.3f})'.format(
                        epoch, i, len(train_loader)-1, batch_time=batch_time,
                        data_time=data_time, loss=losses, mae_errors=mae_errors)
                    )
                else:
                    print('Epoch: [{0}][{1}/{2}]\t'
                        'Time {batch_time.val:.3f} ({batch_time.avg:.3f})\t'
                        'Data {data_time.val:.3f} ({data_time.avg:.3f})\t'
                        'Loss {loss.val:.4f} ({loss.avg:.4f})\t'
                        'Accu {accu.val:.3f} ({accu.avg:.3f})\t'
                        'Precision {prec.val:.3f} ({prec.avg:.3f})\t'
                        'Recall {recall.val:.3f} ({recall.avg:.3f})\t'
                        'F1 {f1.val:.3f} ({f1.avg:.3f})'.format(
                    epoch, i, len(train_loader)-1, batch_time=batch_time,
                    data_time=data_time, loss=losses, accu=accuracies,
                    prec=precisions, recall=recalls, f1=fscores))

            if 'df_train' in kwargs:
                if args.task == 'regression':
                    # Append the log to the list

                    logs = [epoch, i, batch_time.val, batch_time.avg,
                                data_time.val, data_time.avg, losses.val.item(), losses.avg.item(),
                                mae_errors.val.item(), mae_errors.avg.item()]
                else:
                    # Append the log to the list
                    logs = [epoch, i, batch_time.val, batch_time.avg,
                                data_time.val, data_time.avg, losses.val, losses.avg,
                                accuracies.val, accuracies.avg,
                                precisions.val, precisions.avg,
                                recalls.val, recalls.avg,
                                fscores.val, fscores.avg]
            
            if 'df_train' in kwargs:
                # Add the new row to the train DataFrame
                df_train = kwargs['df_train']
                df_train.loc[len(kwargs['df_train'].index)] = logs

            if i == len(train_loader) -1:
                return df_train

        for i, (input, target, batch_cif_ids) in enumerate(val_loader):
            if args.cuda:
                with torch.no_grad():
                    input_var = (Variable(input[0].cuda(non_blocking=True)),

                                [crys_idx.cuda(non_blocking=True) for crys_idx in input[3]])
            else:
                with torch.no_grad():
                    input_var = (Variable(input[0]),
                                Variable(input[1]),
                                input[2],
                                input[3])
                    
            if args.task == 'regression':
                target_normed = normalizer['normalizer'].log_scale(target)
            else:
                target_normed = target[:,-1].view(-1).long()
                additional_features = target[:, :-1]
                additional_features = normalizer.norm(additional_features)

            if args.cuda:
                if args.task == 'regression':
                    additional_features = target_normed[:, :-args.Target_num].cuda(non_blocking=True)
                    target_var = Variable(target_normed[:,-args.Target_num:].cuda(non_blocking=True))
                else:
                    target_var = Variable(target_normed.cuda(non_blocking=True))
                    target = target_normed.cuda(non_blocking=True)

            else:
                if args.task == 'regression':
                    additional_features = target_normed[:, :-args.Target_num]
                    target_var = Variable(target_normed[:,-args.Target_num:])
                else:
                    target = target_normed
                    target_var = Variable(target_normed)

            # compute output
            output = model(*input_var,additional_features.cuda() if args.cuda else additional_features)
            loss = criterion(output, target_var)

            # measure accuracy and record loss
            if args.task == 'regression':
                mae_error = mae(normalizer['target_normalizer'].log_descale(output.data.cpu()), target[:,-args.Target_num:])
                losses.update(loss.data.cpu().item(), target.size(0))
                mae_errors.update(mae_error, target.size(0))

                if 'df_test' in kwargs:
                    # Append the log to the list
                    logs = [i, batch_time.val, batch_time.avg, losses.val, losses.avg,
                            mae_errors.val.item(), mae_errors.avg.item()]
                if test:
                    test_pred = normalizer['target_normalizer'].log_descale(output.data.cpu())
                    test_target = target[:,-args.Target_num:]
                    test_preds += test_pred.tolist()
                    test_targets += test_target.tolist()
                    test_cif_ids += batch_cif_ids
            else:
                accuracy, precision, recall, fscore = \
                    class_eval(output.data.cpu(), target)
                losses.update(loss.data.cpu().item(), target.size(0))
                accuracies.update(accuracy, target.size(0))
                precisions.update(precision, target.size(0))
                recalls.update(recall, target.size(0))
                fscores.update(fscore, target.size(0))

                if 'df_test' in kwargs:
                    logs = [i, batch_time.val, batch_time.avg,
                            losses.val, losses.avg,
                            accuracies.val, accuracies.avg,
                            precisions.val, precisions.avg,
                            recalls.val, recalls.avg,
                            fscores.val, fscores.avg]
                
                # Gather the model predictions
                if test:
                        test_pred = torch.exp(output.data.cpu())
                        test_target = target_var
                        assert test_pred.shape[1] == args.Class_num 
                        test_pred = torch.argmax(test_pred,axis=1)
                        test_preds += test_pred[:].tolist()
                        test_targets += test_target.view(-1).tolist() 
                        test_cif_ids += batch_cif_ids

            # measure elapsed time
            batch_time.update(time.time() - end)
            end = time.time()

            if i % args.print_freq == 0:
                if args.task == 'regression':
                    print('Test: [{0}/{1}]\t'
                        'Time {batch_time.val:.3f} ({batch_time.avg:.3f})\t'
                        'Loss {loss.val:.4f} ({loss.avg:.4f})\t'
                        'MAE {mae_errors.val:.3f} ({mae_errors.avg:.3f})'.format(
                        i, len(val_loader)-1, batch_time=batch_time, loss=losses,
                        mae_errors=mae_errors))
                else:
                    print('Test:[{0}/{1}]\t'
                      'Time {batch_time.val:.3f} ({batch_time.avg:.3f})\t'
                      'Loss {loss.val:.4f} ({loss.avg:.4f})\t'
                      'Accu {accu.val:.3f} ({accu.avg:.3f})\t'
                      'Precision {prec.val:.3f} ({prec.avg:.3f})\t'
                      'Recall {recall.val:.3f} ({recall.avg:.3f})\t'
                      'F1 {f1.val:.3f} ({f1.avg:.3f})\t'.format(
                    i, len(val_loader)-1, batch_time=batch_time, loss=losses,
                    accu=accuracies, prec=precisions, recall=recalls,
                    f1=fscores))

            if 'df_test' in kwargs:
                # Add the new row to the train DataFrame
                df_test.loc[len(df_test.index)] = logs

        if test:
            star_label = '**'
            import csv

            with open(filename, 'w') as f:
                writer = csv.writer(f)
                for cif_id, target, pred in zip(test_cif_ids, test_targets,
                                                test_preds):
                    writer.writerow((cif_id, target, pred))
        else:
            star_label = '*'

        if args.task == 'regression':
            print(' {star} MAE {mae_errors.avg:.3f}'.format(star=star_label,
                                                            mae_errors=mae_errors))
            return mae_errors.avg, df_test if 'df_test' in kwargs else None
        else:
            print(' {star} Accuracy {accu.avg:.3f}'.format(star=star_label,
                                                accu=accuracies))
            return accuracies.avg, df_test if 'df_test' in kwargs else None



# Determines other performance metricies
def class_eval(prediction, target):
    prediction = np.exp(prediction.numpy())
    target = target.cpu().numpy()
    pred_label = np.argmax(prediction, axis=1)
    target_label = np.squeeze(target)
    if not target_label.shape:
        target_label = np.asarray([target_label])
    precision, recall, fscore, _ = metrics.precision_recall_fscore_support(
        target_label, pred_label, average='weighted')
    accuracy = metrics.accuracy_score(target_label, pred_label)
    return accuracy, precision, recall, fscore

class Normalizer(object):
    """Normalize a Tensor column by column and restore it later. """

    def __init__(self, tensor):
        """tensor is taken as a sample to calculate the mean and std"""
        self.mean = torch.mean(tensor, dim=0)
        self.std = torch.std(tensor, dim=0)

    def norm(self, tensor):
        return (tensor - self.mean) / self.std

    def denorm(self, normed_tensor):
        return normed_tensor * self.std + self.mean

    def state_dict(self):
        return {'mean': self.mean,
                'std': self.std}

    def load_state_dict(self, state_dict):
        self.mean = state_dict['mean']
        self.std = state_dict['std']

    # Log scaling method
    def log_scale(self, tensor):
        # Add 1 to avoid taking log of 0
        return torch.log(tensor + 1)

    # Log descaling method
    def log_descale(self, log_scaled_tensor):
        # Use exp to reverse the log transformation
        return torch.exp(log_scaled_tensor) - 1


class AverageMeter(object):
    """Computes and stores the average and current value"""

    def __init__(self):
        self.reset()

    def reset(self):
        self.val = 0
        self.avg = 0
        self.sum = 0
        self.count = 0

    def update(self, val, n=1):
        self.val = val
        self.sum += val * n
        self.count += n
        self.avg = self.sum / self.count


def mae(prediction, target):
    """
    Computes the mean absolute error between prediction and target

    Parameters
    ----------

    prediction: torch.Tensor (N, 1)
    target: torch.Tensor (N, 1)
    """
    return torch.mean(torch.abs(target - prediction))
